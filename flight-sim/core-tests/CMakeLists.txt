cmake_minimum_required(VERSION 3.16)
project(KanafCoreTests CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
# Same flags spirit as Unreal: no exceptions/RTTI reliance in core code.
if(MSVC)
  add_compile_options(/W4 /utf-8)
else()
  add_compile_options(-Wall -Wextra -Wno-unused-parameter -fno-exceptions -fno-rtti)
endif()
set(CORE_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../KanafSim/Source/KanafCore)
file(GLOB CORE_SRC ${CORE_DIR}/Private/*.cpp)
list(FILTER CORE_SRC EXCLUDE REGEX "KanafCoreModule\\.cpp$")
add_library(kanaf_core STATIC ${CORE_SRC})
target_include_directories(kanaf_core PUBLIC ${CORE_DIR}/Public)

set(DATA_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../KanafSim/Content/KanafData)
add_executable(kanaf_tests tests/test_main.cpp)
target_link_libraries(kanaf_tests kanaf_core)
target_compile_definitions(kanaf_tests PRIVATE KANAF_DATA_DIR="${DATA_DIR}")

add_executable(kanaf_sim_cli tests/sim_cli.cpp)
target_link_libraries(kanaf_sim_cli kanaf_core)
target_compile_definitions(kanaf_sim_cli PRIVATE KANAF_DATA_DIR="${DATA_DIR}")

enable_testing()
add_test(NAME kanaf_tests COMMAND kanaf_tests)
